Aquí tienes el script SQL completo de la base de datos ordenado de manera impecable, junto con el poblado inicial exclusivo para el catálogo de **Actos de Servicio**, **Retos de Desconexión**, **Reflexiones Diarias** y la estructura de **Recompensas Reales** (sin datos de usuarios, chats, logs ni ubicaciones).

Para evitar que la inserción de recompensas de ejemplo falle debido a la restricción de clave foránea (`partnership_id`), se incluye un ejemplo comentado y limpio de cómo poblar las recompensas asociadas a una relación activa.

### Script SQL Ordenado y Poblado de Catálogos

```sql
-- =========================================================================
-- PARTE 1: CREACIÓN DE TABLAS EN ORDEN DE DEPENDENCIAS
-- =========================================================================

-- 1. Tabla de perfiles de usuario (Depende de auth.users)
CREATE TABLE IF NOT EXISTS public.profiles (
  id uuid NOT NULL,
  full_name text,
  avatar_url text,
  love_language text,
  updated_at timestamp with time zone DEFAULT now(),
  total_points integer NOT NULL DEFAULT 0,
  birth_date date,
  preferences jsonb DEFAULT '{}'::jsonb,
  fcm_token text,
  current_month_percentage numeric DEFAULT 0,
  current_month_snapshot_at timestamp with time zone,
  last_recorded_month integer,
  last_recorded_year integer,
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id) ON DELETE CASCADE
);

-- 2. Tabla de reflexiones diarias (Independiente)
CREATE TABLE IF NOT EXISTS public.daily_reflections (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  phrase text NOT NULL,
  author text DEFAULT 'Anónimo'::text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT daily_reflections_pkey PRIMARY KEY (id)
);

-- 3. Tabla de emparejamiento / relaciones (Depende de public.profiles)
CREATE TABLE IF NOT EXISTS public.partnerships (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user1_id uuid UNIQUE,
  user2_id uuid UNIQUE,
  pairing_token text UNIQUE,
  status text DEFAULT 'pending'::text,
  anniversary_date date,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT partnerships_pkey PRIMARY KEY (id),
  CONSTRAINT partnerships_user1_id_fkey FOREIGN KEY (user1_id) REFERENCES public.profiles(id) ON DELETE SET NULL,
  CONSTRAINT partnerships_user2_id_fkey FOREIGN KEY (user2_id) REFERENCES public.profiles(id) ON DELETE SET NULL
);

-- 4. Tabla de ubicaciones (Depende de public.profiles)
CREATE TABLE IF NOT EXISTS public.locations (
  user_id uuid NOT NULL,
  latitude double precision,
  longitude double precision,
  privacy_enabled boolean DEFAULT true,
  last_update timestamp with time zone DEFAULT now(),
  CONSTRAINT locations_pkey PRIMARY KEY (user_id),
  CONSTRAINT locations_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id) ON DELETE CASCADE
);

-- 5. Tabla de geozonas (Depende de public.partnerships)
CREATE TABLE IF NOT EXISTS public.geozones (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  partnership_id uuid,
  name text,
  latitude double precision,
  longitude double precision,
  radius integer DEFAULT 50,
  CONSTRAINT geozones_pkey PRIMARY KEY (id),
  CONSTRAINT geozones_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE CASCADE
);

-- 6. Tabla de recuerdos (Depende de public.partnerships)
CREATE TABLE IF NOT EXISTS public.memories (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  partnership_id uuid,
  file_url text,
  emotional_score double precision,
  location_name text,
  created_at timestamp with time zone DEFAULT now(),
  file_name text,
  user_id uuid,
  CONSTRAINT memories_pkey PRIMARY KEY (id),
  CONSTRAINT memories_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE CASCADE
);

-- 7. Tabla de alertas SOS (Depende de public.profiles)
CREATE TABLE IF NOT EXISTS public.sos_alerts (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  latitude double precision,
  longitude double precision,
  audio_url text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT sos_alerts_pkey PRIMARY KEY (id),
  CONSTRAINT sos_alerts_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id) ON DELETE CASCADE
);

-- 8. Catálogo global de actividades (Depende de public.partnerships opcionalmente)
CREATE TABLE IF NOT EXISTS public.activity_catalog (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name text NOT NULL,
  category text NOT NULL,
  default_points integer NOT NULL,
  activity_type text DEFAULT 'ROUTINE'::text CHECK (activity_type = ANY (ARRAY['ROUTINE'::text, 'CHALLENGE'::text])),
  subcategory text,
  description text,
  partnership_id uuid,
  CONSTRAINT activity_catalog_pkey PRIMARY KEY (id),
  CONSTRAINT activity_catalog_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE SET NULL
);

-- 9. Registro histórico de acciones realizadas (Depende de public.profiles y public.activity_catalog)
CREATE TABLE IF NOT EXISTS public.user_actions_log (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  action_id uuid,
  manual_description text,
  points_earned integer NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  status text DEFAULT 'CONFIRMED'::text,
  validated_by uuid,
  partnership_id uuid,
  CONSTRAINT user_actions_log_pkey PRIMARY KEY (id),
  CONSTRAINT user_actions_log_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id) ON DELETE CASCADE,
  CONSTRAINT user_actions_log_action_id_fkey FOREIGN KEY (action_id) REFERENCES public.activity_catalog(id) ON DELETE SET NULL,
  CONSTRAINT user_actions_log_validated_by_fkey FOREIGN KEY (validated_by) REFERENCES public.profiles(id) ON DELETE SET NULL
);

-- 10. Libro de transacciones de puntos (Depende de partnerships, profiles y activity_catalog)
CREATE TABLE IF NOT EXISTS public.points_ledger (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  partnership_id uuid,
  user_id uuid,
  points integer NOT NULL,
  ai_validated boolean DEFAULT false,
  created_at timestamp with time zone DEFAULT now(),
  act_id uuid,
  CONSTRAINT points_ledger_pkey PRIMARY KEY (id),
  CONSTRAINT points_ledger_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE CASCADE,
  CONSTRAINT points_ledger_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id) ON DELETE CASCADE,
  CONSTRAINT points_ledger_act_id_fkey FOREIGN KEY (act_id) REFERENCES public.activity_catalog(id) ON DELETE SET NULL
);

-- 11. Tabla de recompensas reales intercambiables (Depende de public.partnerships)
CREATE TABLE IF NOT EXISTS public.real_rewards (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  partnership_id uuid,
  description text NOT NULL,
  point_cost integer NOT NULL,
  is_redeemed boolean DEFAULT false,
  CONSTRAINT real_rewards_pkey PRIMARY KEY (id),
  CONSTRAINT real_rewards_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE CASCADE
);

-- 12. Salas de chat (Depende de public.partnerships y auth.users)
CREATE TABLE IF NOT EXISTS public.chat_rooms (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  partnership_id uuid,
  room_type text NOT NULL CHECK (room_type = ANY (ARRAY['COUPLE'::text, 'PRIVATE_AI'::text, 'GROUP_AI'::text])),
  created_at timestamp with time zone DEFAULT now(),
  title text DEFAULT 'Nueva Conversación'::text,
  user_id uuid,
  CONSTRAINT chat_rooms_pkey PRIMARY KEY (id),
  CONSTRAINT chat_rooms_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE CASCADE,
  CONSTRAINT chat_rooms_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE SET NULL
);

-- 13. Mensajes en las salas de chat (Depende de public.chat_rooms y public.profiles)
CREATE TABLE IF NOT EXISTS public.chat_messages (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  room_id uuid NOT NULL,
  sender_id uuid,
  sender_type text NOT NULL CHECK (sender_type = ANY (ARRAY['USER'::text, 'AI'::text])),
  message text NOT NULL,
  metadata jsonb DEFAULT '{}'::jsonb,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT chat_messages_pkey PRIMARY KEY (id),
  CONSTRAINT chat_messages_room_id_fkey FOREIGN KEY (room_id) REFERENCES public.chat_rooms(id) ON DELETE CASCADE,
  CONSTRAINT chat_messages_sender_id_fkey FOREIGN KEY (sender_id) REFERENCES public.profiles(id) ON DELETE SET NULL
);

-- 14. Puntos pendientes (Versión española - Depende de auth.users)
CREATE TABLE IF NOT EXISTS public.puntos_pendientes (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  emisor_id uuid,
  receptor_id uuid,
  actividad_nombre text NOT NULL,
  puntos_valor integer NOT NULL,
  estado text DEFAULT 'pendiente'::text CHECK (estado = ANY (ARRAY['pendiente'::text, 'aprobado'::text, 'rechazado'::text])),
  CONSTRAINT puntos_pendientes_pkey PRIMARY KEY (id),
  CONSTRAINT puntos_pendientes_emisor_id_fkey FOREIGN KEY (emisor_id) REFERENCES auth.users(id) ON DELETE CASCADE,
  CONSTRAINT puntos_pendientes_receptor_id_fkey FOREIGN KEY (receptor_id) REFERENCES auth.users(id) ON DELETE CASCADE
);

-- 15. Puntos pendientes (Versión inglesa/core - Depende de auth.users)
CREATE TABLE IF NOT EXISTS public.pending_points (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  sender_id uuid NOT NULL,
  receiver_id uuid NOT NULL,
  activity_name text NOT NULL,
  points_value integer NOT NULL,
  status text DEFAULT 'pending'::text CHECK (status = ANY (ARRAY['pending'::text, 'approved'::text, 'rejected'::text])),
  description text,
  CONSTRAINT pending_points_pkey PRIMARY KEY (id),
  CONSTRAINT pending_points_sender_id_fkey FOREIGN KEY (sender_id) REFERENCES auth.users(id) ON DELETE CASCADE,
  CONSTRAINT pending_points_receiver_id_fkey FOREIGN KEY (receiver_id) REFERENCES auth.users(id) ON DELETE CASCADE
);

-- 16. Historial de afinidad mensual acumulada (Depende de public.partnerships y auth.users)
CREATE TABLE IF NOT EXISTS public.monthly_affinity (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  partnership_id uuid,
  user_id uuid,
  year integer NOT NULL,
  month integer NOT NULL,
  percentage numeric NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT monthly_affinity_pkey PRIMARY KEY (id),
  CONSTRAINT monthly_affinity_partnership_id_fkey FOREIGN KEY (partnership_id) REFERENCES public.partnerships(id) ON DELETE CASCADE,
  CONSTRAINT monthly_affinity_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE
);
```
